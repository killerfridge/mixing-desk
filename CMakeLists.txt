cmake_minimum_required(VERSION 3.24)
project(MixingDesk LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/W4 /permissive- /utf-8 /EHsc)
  add_compile_definitions(_USE_MATH_DEFINES)
endif()
if(WIN32 AND NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The Windows beta requires x64")
endif()
include(FetchContent)
FetchContent_Declare(json URL https://codeload.github.com/nlohmann/json/tar.gz/refs/tags/v3.11.3
  URL_HASH SHA256=0d8ef5af7f9794e3263480193c491549b2ba6cc74bb018906202ada498a79406)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
set(LIBSAMPLERATE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LIBSAMPLERATE_INSTALL OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
FetchContent_Declare(samplerate URL https://codeload.github.com/libsndfile/libsamplerate/tar.gz/refs/tags/0.2.2
  URL_HASH SHA256=16e881487f184250deb4fcb60432d7556ab12cb58caea71ef23960aec6c0405a)
FetchContent_MakeAvailable(json samplerate)
find_package(Threads REQUIRED)
add_library(DeskEngine STATIC Sources/DeskAudio/Engine.cpp)
target_include_directories(DeskEngine PUBLIC Sources/DeskAudio Sources/DeskAudio/VST3SDK)
target_link_libraries(DeskEngine PUBLIC Threads::Threads)
add_library(DeskWindows SHARED Windows/Native/Bridge.cpp Windows/Native/Session.cpp)
if(WIN32)
  set_target_properties(DeskWindows PROPERTIES PREFIX "")
endif()
target_link_libraries(DeskWindows PRIVATE DeskEngine nlohmann_json::nlohmann_json samplerate)
target_compile_definitions(DeskWindows PRIVATE MDW_BUILD)
if(WIN32)
  target_sources(DeskWindows PRIVATE Windows/Native/WasapiHost.cpp Windows/Native/VstHost.cpp Sources/DeskAudio/VST3Identifiers.cpp Sources/DeskAudio/VST3SDK/pluginterfaces/base/funknown.cpp)
  target_compile_definitions(DeskWindows PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(DeskWindows PRIVATE ole32 uuid avrt propsys mmdevapi)
  add_executable(DeskPluginScanner Windows/Native/PluginScanner.cpp Windows/Native/VstHost.cpp Windows/Native/Session.cpp Sources/DeskAudio/VST3Identifiers.cpp Sources/DeskAudio/VST3SDK/pluginterfaces/base/funknown.cpp)
  target_include_directories(DeskPluginScanner PRIVATE Sources/DeskAudio)
  target_compile_definitions(DeskPluginScanner PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(DeskPluginScanner PRIVATE DeskEngine nlohmann_json::nlohmann_json ole32 uuid)
  if(MINGW)
    target_link_options(DeskPluginScanner PRIVATE -municode)
  endif()
  add_library(DeskVstFixture MODULE Windows/Tests/VstFixture.cpp Sources/DeskAudio/VST3Identifiers.cpp Sources/DeskAudio/VST3SDK/pluginterfaces/base/funknown.cpp)
  target_link_libraries(DeskVstFixture PRIVATE DeskEngine ole32)
  set_target_properties(DeskVstFixture PROPERTIES PREFIX "" SUFFIX ".vst3")
  add_executable(DeskVstTests Windows/Tests/VstTests.cpp Windows/Native/VstHost.cpp Windows/Native/Session.cpp Sources/DeskAudio/VST3Identifiers.cpp Sources/DeskAudio/VST3SDK/pluginterfaces/base/funknown.cpp)
  target_compile_definitions(DeskVstTests PRIVATE MDW_PLUGIN_TESTS NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(DeskVstTests PRIVATE DeskEngine nlohmann_json::nlohmann_json ole32 uuid)
  foreach(t DeskVstFixture DeskVstTests)
    if(MSVC)
      target_compile_options(${t} PRIVATE /UNDEBUG)
    else()
      target_compile_options(${t} PRIVATE -UNDEBUG)
    endif()
  endforeach()
endif()
enable_testing()
if(WIN32)
  add_test(NAME DeskVstTests COMMAND DeskVstTests $<TARGET_FILE:DeskVstFixture>)
  add_test(NAME DeskVstScanner COMMAND DeskPluginScanner $<TARGET_FILE:DeskVstFixture> ${CMAKE_BINARY_DIR}/scanner-result.json)
endif()
add_executable(DeskEngineTests Tests/EngineTests.cpp)
target_link_libraries(DeskEngineTests PRIVATE DeskEngine)
add_executable(DeskWindowsTests Windows/Tests/NativeTests.cpp Windows/Native/Session.cpp)
target_link_libraries(DeskWindowsTests PRIVATE DeskEngine nlohmann_json::nlohmann_json samplerate)
foreach(t DeskEngineTests DeskWindowsTests)
  if(MSVC)
    target_compile_options(${t} PRIVATE /UNDEBUG)
    target_link_options(${t} PRIVATE /STACK:8388608)
  else()
    target_compile_options(${t} PRIVATE -UNDEBUG)
    if(MINGW)
      target_link_options(${t} PRIVATE -Wl,--stack,8388608)
    endif()
  endif()
  add_test(NAME ${t} COMMAND ${t})
endforeach()
